  /* ── arrow keys drive whichever slider is on screen ── */
  (function(){
    var list=root.__sliders||[]; if(!list.length) return;
    function pick(){
      var mid=window.innerHeight/2, best=null, bd=Infinity;
      list.forEach(function(s){ var r=s.el.getBoundingClientRect(); if(r.bottom<0||r.top>window.innerHeight) return;
        var d=Math.abs((r.top+r.bottom)/2-mid); if(d<bd){bd=d;best=s;} });
      return best;
    }
    document.addEventListener('keydown',function(e){
      if(e.key!=='ArrowLeft'&&e.key!=='ArrowRight') return;
      var t=e.target, tag=(t.tagName||'').toLowerCase();
      if(tag==='input'||tag==='textarea'||tag==='select'||t.isContentEditable) return;
      /* a focused slider wins; otherwise the one nearest the middle of the screen */
      var s=list.find(function(x){return x.el===t||x.el.contains(t);}) || pick();
      if(!s) return;
      e.preventDefault(); s.step(e.key==='ArrowRight'?1:-1);
    });
  })();
